Enum membership_tier {
  PRIVATE [note: '이등병 (기본값)']
  PFC [note: '일병']
  CORPORAL [note: '상병']
  SERGEANT [note: '병장']
}

Table users {
  id bigint [pk, increment]
  login_id varchar [not null, unique, note: '가상 회원 식별자']
  name varchar [not null, note: 'PII - 마스킹 대상']
  phone varchar [not null, note: 'PII - 마스킹 대상']
  email varchar [not null, note: 'PII - 마스킹 대상']
  membership_tier membership_tier [not null, default: 'PRIVATE', note: 'orders 집계 기반 매월 1일 배치 자동 산정 (FR-MBR-001,002). 최초 더미데이터 생성 시점 값은 orders 더미데이터로부터 역산']
  tier_calculated_at datetime [note: '마지막 계급 재산정 배치 실행 시각']
  created_at datetime [not null, default: `now()`]
}

Table orders {
  id bigint [pk, increment]
  user_id bigint [not null]
  order_amount decimal [not null, note: '상품 금액(할인 전)']
  paid_amount decimal [not null, note: '결제 금액(할인 후) — 계급 집계 10,000원 기준 컬럼 (FR-MBR-003, 팀 확정: 결제금액 기준)']
  status varchar [not null, note: 'ENUM: COMPLETED, CANCELED, REFUNDED']
  ordered_at datetime [not null]
  completed_at datetime [note: '배달완료 시각. 계급 집계는 status=COMPLETED AND completed_at 기준 캘린더 월로 산정']
  coupon_issue_id bigint [note: '[V8 추가] 적용된 쿠폰 발급 건. nullable(쿠폰 미적용 주문도 있음) — 주문취소 시 쿠폰 상태전이(ISSUED 복귀) 정합성 확인용']
  created_at datetime [not null, default: `now()`]

  indexes {
    (user_id, status, completed_at) [name: 'idx_orders_tier_calc']
  }
}

Table membership_tier_log {
  id bigint [pk, increment]
  user_id bigint [not null]
  from_tier membership_tier
  to_tier membership_tier [not null]
  order_count int [not null, note: '해당 배치 산정에 집계된 완료 주문 건수']
  calculated_at datetime [not null, default: `now()`, note: '배치 실행 시각(매월 1일)']
}

Table campaign {
  id bigint [pk, increment]
  name varchar [not null]
  total_stock int [not null, note: '최초 재고, 불변']
  remaining_stock int [not null, note: '[2026-08-20 V4 이후 의미 변경] 더 이상 재고 판단의 원본이 아님 — 진짜 재고는 campaign_stock_shard 합계이고, 이 컬럼은 관리자 CRUD·검증쿼리(b) 등 기존 코드가 계속 읽을 수 있도록 스냅샷 리스너/재동기화 잡이 그 합계를 사후 복사해두는 표시값']
  open_at datetime [note: '예약 오픈 (선택 확장)']
  close_at datetime
  status varchar [not null, note: 'ENUM: READY, OPEN, CLOSED']
  min_membership_tier membership_tier [note: 'NULL이면 전 회원 대상, 설정 시 회원 전용 쿠폰']
  campaign_type varchar [not null, default: 'FCFS', note: '[V5, 2026-08-19 추가] ENUM: FCFS(선착순), MEMBERSHIP_BENEFIT(계급별 월간 혜택)']
  version bigint [not null, default: 0, note: '@Version, 낙관적 락 옵션용']
}

Table campaign_stock_shard {
  id bigint [pk, increment]
  campaign_id bigint [not null]
  shard_index int [not null]
  remaining_stock int [not null, note: '이 샤드 몫의 잔여 재고 — campaign.remaining_stock을 대체하는 진짜 재고 원본(V4, 재고 샤딩)']
  capacity int [not null, note: '샤드 생성 시점 값으로 고정. sum(capacity)=campaign.total_stock 불변식']

  indexes {
    (campaign_id, shard_index) [unique, name: 'uk_campaign_shard']
  }
}

Table coupon {
  id bigint [pk, increment]
  campaign_id bigint [not null]
  discount_type varchar [not null, note: 'ENUM: FIXED, RATE']
  discount_value decimal [not null]
  min_order_amount decimal [note: '최소 주문금액 조건부 할인, NULL이면 조건 없음']
  max_discount_amount decimal [note: '정률 할인 최대 한도, NULL이면 한도 없음 (FR-CPN-002, 02_기능명세서_개정본 15절)']
  valid_hours int [not null, note: '발급 시점(issued_at)으로부터 유효 시간(시간 단위). 하루 미만(당일 한정 등) 쿠폰도 표현하기 위해 일 단위 대신 시간 단위 사용. 반드시 1 이상 — 0이면 발급 즉시 만료된 죽은 쿠폰이 나감']
}

Table coupon_issue {
  id bigint [pk, increment]
  campaign_id bigint [not null]
  user_id bigint [not null]
  coupon_code varchar [not null, unique, note: '발급된 실물 쿠폰 식별자']
  discount_type varchar [not null, note: 'ENUM: FIXED, RATE — 발급 시점 스냅샷']
  discount_value decimal [not null, note: '발급 시점 계급별 할인율/금액 스냅샷(04_아키텍처.md 6.1절 TierDiscountPolicy). 이후 계급 변동과 무관하게 불변']
  max_discount_amount decimal [note: '발급 시점 한도 스냅샷. discount_value와 동일 원칙 (FR-CPN-002, 02_기능명세서_개정본 15절)']
  issued_membership_tier membership_tier [not null, note: '발급 시점 유저 계급 스냅샷. 검증쿼리(d)가 users.membership_tier(변동값) 대신 이 값으로 판정해야 계급 재산정 후 false positive가 없음']
  status varchar [not null, note: 'ENUM: ISSUED, USED, CANCELED, EXPIRED']
  idempotency_key varchar [not null, unique, note: '발급 요청 중복 방지 키']
  issued_at datetime [not null]
  valid_until datetime [not null, note: '만료 마감 시점 스냅샷 = issued_at + coupon.valid_hours, 발급 시점에 고정. coupon.valid_hours가 나중에 바뀌어도 이미 발급된 건은 불변 (FR-CPS-003)']
  used_at datetime
  canceled_at datetime
  expired_at datetime [note: '실제로 EXPIRED 상태로 전이 처리된 시각(이력). valid_until은 마감선, expired_at은 처리 시각으로 서로 다른 값']
  version bigint [not null, default: 0, note: '@Version, 상태전이 낙관적 락']
  created_at datetime [not null, default: `now()`]
  updated_at datetime [not null, default: `now()`]

  indexes {
    (campaign_id, user_id) [unique, name: 'uk_campaign_user']
    campaign_id [name: 'idx_ci_campaign']
    (status, valid_until) [name: 'idx_ci_status_valid_until', note: '만료 배치가 WHERE status=ISSUED AND valid_until < NOW()로 스캔할 때 사용. 없으면 300만 건 풀스캔']
  }
}

Table coupon_state_log {
  id bigint [pk, increment]
  coupon_issue_id bigint [not null]
  from_status varchar
  to_status varchar [not null]
  request_id varchar [not null, note: '상태변경 요청 idempotency key']
  reason varchar [not null, default: 'UNKNOWN', note: '[V7 추가] 왜 상태가 바뀌었는지 사유. 기존 로우는 소급판정 불가라 UNKNOWN, 새 코드는 UNKNOWN 안 씀']
  created_at datetime [not null, default: `now()`]

  indexes {
    request_id [unique, name: 'uk_state_log_request']
    (coupon_issue_id, id) [name: 'idx_state_log_coupon_issue']
  }
}

// 정합성 검증 배치 결과 (05_시스템설계.txt 1.2절, ConsistencyVerificationJob)
// job_execution_id는 Spring Batch 자체 테이블 BATCH_JOB_EXECUTION 참조 — 이 ERD엔 Spring Batch
// 인프라 테이블(BATCH_JOB_EXECUTION 등)은 모델링하지 않으므로 Ref 없이 컬럼만 표기.
// =========================================================

Table verification_result {
  id bigint [pk, increment]
  job_execution_id bigint [not null, note: 'FK -> BATCH_JOB_EXECUTION.JOB_EXECUTION_ID (Spring Batch 인프라 테이블, 미모델링)']
  check_type varchar [not null, note: '위반 검증쿼리 종류 (a_초과발급/b_카운터불일치/c1~c3/d_등급위반/e_등급-주문불일치)']
  reference_id varchar [not null, note: '위반이 발견된 대상의 식별자(캠페인ID, 발급건ID 등)']
  detail text [not null]
  detected_at datetime [not null]

  indexes {
    job_execution_id [name: 'idx_verification_result_job']
    check_type [name: 'idx_verification_result_check_type']
  }
}

Table verification_report {
  id bigint [pk, increment]
  job_execution_id bigint [not null, note: 'FK -> BATCH_JOB_EXECUTION.JOB_EXECUTION_ID (미모델링)']
  job_name varchar [not null]
  started_at datetime
  ended_at datetime
  duration_ms bigint [not null, default: 0]
  total_violation_count bigint [not null, default: 0]
  status varchar [not null]
  report_file_path varchar [note: 'Notion 리포트 링크 등']
  created_at datetime [not null, default: `now()`]

  indexes {
    job_execution_id [unique, name: 'uk_verification_report_job_execution']
    (job_name, created_at) [name: 'idx_verification_report_job_name_created']
  }
}
